-- DPS-søknadspakker 2 (trading-1, 2026-10-20): åpne ordninger hos Vinmonopolet, Digdir, Entur, Mattilsynet, Flytoget.
-- READ-ONLY, bare CTE-er.
BEGIN;
SET TRANSACTION READ ONLY;
WITH k AS (
  SELECT t.id, t.external_id, t.title, t.deadline, b.name kjoper, t.documents_url, t.qualification_criteria, t.description, t.notice_type
  FROM tenders t JOIN buyers b ON b.id = t.buyer_id
  WHERE t.deadline > now()
    AND b.name ~* '(vinmonopol|digitaliseringsdirektorat|entur|mattilsyn|flytoget)'
    AND (t.title ~* '(dynamisk|DPS|innkjøpsordning|konsulent|utvikl|test)' OR t.description ~* 'dynamisk innkjøpsordning')
)
SELECT 'ORDNING', external_id, kjoper, to_char(deadline,'YYYY-MM-DD HH24:MI'), title, coalesce(documents_url,'-'),
  CASE WHEN jsonb_typeof(qualification_criteria::jsonb)='array' THEN jsonb_array_length(qualification_criteria::jsonb) ELSE 0 END::text
FROM k ORDER BY kjoper, external_id;
-- DPS-søknadspakker 2 (trading-1, 2026-10-20): fulle krav for 12 utvalgte ordninger. READ-ONLY.
BEGIN;
SET TRANSACTION READ ONLY;
WITH ids(eid) AS (VALUES ('2024-105499'),('2024-105583'),('2022-673748'),('2024-106399'),('2025-112751'),('2024-106024'),
  ('2021-320600'),('2021-386923'),('2021-361268'),('2025-103853'),('2023-664176')),
t AS (SELECT t.* FROM tenders t JOIN ids ON ids.eid = t.external_id WHERE t.source = 'doffin')
SELECT 'BESKR', external_id, coalesce(description,'') FROM t
UNION ALL
SELECT 'KRAV', t.external_id, coalesce(e->>'kind','') || ' | ' || coalesce(e->>'code','') || ' | ' || coalesce(e->>'name','') || ' | ' || coalesce(e->>'description','')
FROM t, jsonb_array_elements(CASE WHEN jsonb_typeof(t.qualification_criteria::jsonb)='array' THEN t.qualification_criteria::jsonb ELSE '[]'::jsonb END) e
UNION ALL
SELECT 'RAW', x.external_id, x.s FROM (
  SELECT DISTINCT r.external_id, left(v #>> '{}', 2500) s
  FROM raw_notices r JOIN ids ON ids.eid = r.external_id,
    jsonb_path_query(r.body::jsonb, 'strict $.**') v
  WHERE jsonb_typeof(v) = 'string' AND length(v #>> '{}') > 60
    AND (v #>> '{}') ~* '(kvalifikasjon|referans|omsetning|kreditt|ISO ?9001|ISO ?27001|ISO ?14001|miljø|erfaring|sertifi|qualification|turnover|kompetanse)'
) x
ORDER BY 2, 1;
